3. beam.Resource Lifecycle

pub const DspData = struct {
    overlap_buf: [480]u8,
    samples_processed: u64,
};

pub const DspContext =
    beam.Resource(*DspData, @import("root"), .{});

// Allocation:
pub fn new_context() beam.term {
    var ptr = beam.allocator.create(DspData) catch ...;
    return DspContext.make(ptr, .{}); // -> #Reference<...>
}

// Unpacking in NIF:
pub fn run(res: DspContext) beam.term {
    const ctx = res.unpack(); // returns *DspData
}

Destructor is invoked automatically by BEAM GC when reference is collected.

4. SIMD Vectorization (`@Vector`)

// 16 lanes of 32-bit floats (512-bit / AVX2 / NEON)
const Vec16 = @Vector(16, f32);

var v_i: Vec16 = undefined;
var v_q: Vec16 = undefined;

// Fill lanes...
const mag_sq = (v_i * v_i) + (v_q * v_q);
const mag_vec = @sqrt(mag_sq); // 16 sqrt in 1 clock cycle!

Target-independent: lowers to ymm on x86_64 AVX2, or q0-q31 on ARM64 NEON.

5. Boundary Overlap Buffer Invariant

Problem: ADS-B Mode-S frame = 240 samples (480 bytes). Frames straddling 64 KB boundary are dropped without stitching. 

// Prepend previous chunk tail (480 bytes)
// to current chunk (65,536 bytes):
const total_bytes = ctx.overlap_len + chunk.len;
// Store last 480 bytes of current chunk for next delivery:
const keep = @min(480, chunk.len);
@memcpy(ctx.overlap_buf[0..keep], chunk[chunk.len - keep .. chunk.len]);
ctx.overlap_len = keep;

6. Schedulers & 1ms Rule

  • Standard NIF: Must complete in ≤ 1 millisecond. Never block BEAM scheduler threads!
  • Dirty Schedulers: For operations taking > 1ms, specify dirty: :cpu or dirty: :io: 
    // In Zigler options:
use Zig,
  otp_app: :sdr_scout,
  nifs: [process_huge_chunk: [dirty: :cpu]]
